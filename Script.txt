-- CONFIG

local CONFIG = {
	-- Wheelie
	WheelieAngle    = 55,    -- angle cible (degrés)
	WheelieKp       = 40,
	WheelieKd       = 8,
	WheelieStrength = 1,     -- multiplicateur global

	-- Auto ride
	RideAccel       = 60,    -- puissance max de poussée
	StartSpeed      = 40,    -- vitesse cible au départ (studs/s), réglable avec les boutons - / +
	SpeedMin        = 5,
	SpeedMax        = 100,
	SpeedStep       = 5,     -- pas des boutons - / +
	SpeedGain       = 0.15,  -- réactivité du régulateur (plus haut = rattrape plus vite la vitesse cible)

	-- Auto tourner (cercle sur place)
	CircleRadius     = 15,   -- rayon du cercle (studs) : plus petit = reste plus près du centre
	CircleSpeed      = 25,   -- vitesse max pendant qu'il tourne (trop haut = ça dérape/se renverse)
	TeleportOnTurn   = true, -- Auto Tourner te TP aussi dans la zone de test
	TurnDirection    = 1,    -- 1 = tourne à fond à DROITE, -1 = à gauche
	TurnRate         = 1.6,  -- vitesse de rotation (rad/s) : plus haut = cercle plus serré
	CircleCorrection = 2,    -- force de rappel vers le cercle
	UseSteerFloat    = true, -- braquage via VehicleSeat.SteerFloat (véhicules à roues)
	UseYawImpulse    = true, -- rotation forcée (marche avec n'importe quel véhicule)
	TurnKp           = 30,
	TurnKd           = 20,
	TurnStrength     = 1,    -- augmente si le véhicule tourne trop mollement

	-- Zone de test (locale, invisible pour les autres joueurs)
	TeleportOnWheelie = true,
	ZonePosition      = Vector3.new(0, 5000, 0),
	ZoneSize          = 200,
	RecenterDistance  = 30,    -- la plateforme se recentre sous le véhicule dès qu'il s'écarte de ça
	LoopDistance      = 20000,  -- Boucle Auto : distance (studs) du départ avant le retour automatique
	ResetKey          = Enum.KeyCode.R, -- touche pour revenir au départ manuellement
	ReseatDelay       = 5,     -- attente (s) après une éjection/ragdoll avant de te remettre sur la trott
	UseMountPrompt    = true,  -- pour remonter, simule l'appui sur E (ProximityPrompt de la trott)
	AutoJumpOnEject   = true,  -- saute automatiquement pendant l'attente (au cas où ça annule le ragdoll)
	JumpInterval      = 1,     -- délai (s) entre deux sauts automatiques

	ToggleKey = Enum.KeyCode.RightShift,
}

local state = { AutoRide = false, AutoWheelie = false, AutoTurn = false, AutoLoop = false, AutoReseat = true, AntiAFK = true, TargetSpeed = CONFIG.StartSpeed }
local circleCenter -- centre du cercle (défini quand Auto Tourner s'active)
local lastSeat     -- dernier siège sur lequel tu étais assis (pour remonter après une éjection)

---------------------------------------------------------------------
-- Utilitaires
---------------------------------------------------------------------
local function getSeat()
	local char = player.Character
	local hum = char and char:FindFirstChildOfClass("Humanoid")
	return hum and hum.SeatPart or nil
end

local function zeroVelocity(root)
	for _, d in ipairs(root:GetDescendants()) do
		if d:IsA("BasePart") then
			d.AssemblyLinearVelocity = Vector3.zero
			d.AssemblyAngularVelocity = Vector3.zero
		end
	end
	if root:IsA("BasePart") then
		root.AssemblyLinearVelocity = Vector3.zero
		root.AssemblyAngularVelocity = Vector3.zero
	end
end

local function moveVehicle(seat, targetSeatCF)
	local model = seat:FindFirstAncestorOfClass("Model")
	if model and not model:FindFirstChildOfClass("Humanoid") then
		local offset = seat.CFrame:Inverse() * model:GetPivot()
		model:PivotTo(targetSeatCF * offset)
		zeroVelocity(model)
	else
		seat.CFrame = targetSeatCF
		zeroVelocity(seat)
	end
end

---------------------------------------------------------------------
-- Zone de test locale
---------------------------------------------------------------------
local zone
local inZone = false
local savedSeatCF

local function ensureZone()
	if zone and zone.Parent then return zone end
	zone = Instance.new("Part")
	zone.Name = "KukirinLocalZone"
	zone.Anchored = true
	zone.CanCollide = true
	zone.Size = Vector3.new(CONFIG.ZoneSize, 2, CONFIG.ZoneSize)
	zone.Position = CONFIG.ZonePosition - Vector3.new(0, 1, 0)
	zone.Material = Enum.Material.SmoothPlastic
	zone.Color = Color3.fromRGB(60, 60, 66)
	zone.Parent = Workspace
	return zone
end

local function teleportToZone()
	local seat = getSeat()
	if not seat then return false end
	local z = ensureZone()
	savedSeatCF = seat.CFrame
	z.Position = Vector3.new(CONFIG.ZonePosition.X, CONFIG.ZonePosition.Y - 1, CONFIG.ZonePosition.Z)
	moveVehicle(seat, CFrame.new(CONFIG.ZonePosition + Vector3.new(0, 4, 0)))
	inZone = true
	return true
end

local function returnFromZone()
	if not inZone then return end
	inZone = false
	local seat = getSeat()
	if seat and savedSeatCF then
		moveVehicle(seat, savedSeatCF)
	end
end

---------------------------------------------------------------------
-- Retour au départ (pour refaire le même chemin en boucle)
---------------------------------------------------------------------
local lastReset = 0
local function resetToStart(seatOverride)
	if not inZone or not zone then return end
	local seat = seatOverride or getSeat()
	if not seat then return end
	lastReset = os.clock()
	zone.Position = Vector3.new(CONFIG.ZonePosition.X, CONFIG.ZonePosition.Y - 1, CONFIG.ZonePosition.Z)
	moveVehicle(seat, CFrame.new(CONFIG.ZonePosition + Vector3.new(0, 4, 0)))
end

---------------------------------------------------------------------
-- Auto Ride + Auto Tourner (priorité juste après les inputs)
---------------------------------------------------------------------
RunService:BindToRenderStep("KukirinAutoRide", Enum.RenderPriority.Input.Value + 1, function(dt)
	local seat = getSeat()
	if not seat then return end
	if not (state.AutoRide or state.AutoTurn) then return end

	-- Avance en régulant la vitesse (régulateur) : accélère sous la cible, relâche au-dessus
	local flatLook = Vector3.new(seat.CFrame.LookVector.X, 0, seat.CFrame.LookVector.Z)
	if flatLook.Magnitude > 0.05 then
		flatLook = flatLook.Unit
		local target = state.TargetSpeed
		if state.AutoTurn then
			target = math.min(target, CONFIG.CircleSpeed) -- ralentit automatiquement dans les virages
		end
		local speed = seat.AssemblyLinearVelocity:Dot(flatLook)
		local control = math.clamp((target - speed) * CONFIG.SpeedGain, -0.5, 1)

		if seat:IsA("VehicleSeat") then
			seat.ThrottleFloat = math.max(control, 0) -- jamais de marche arrière
		end
		seat:ApplyImpulse(flatLook * seat.AssemblyMass * control * CONFIG.RideAccel * dt)
	end

	-- Tourne à fond en continu (droite par défaut)
	if state.AutoTurn then
		local dir = CONFIG.TurnDirection -- 1 = droite, -1 = gauche
		if CONFIG.UseSteerFloat and seat:IsA("VehicleSeat") then
			seat.SteerFloat = dir -- braquage maximum
		end
		if CONFIG.UseYawImpulse then
			local yawVel = seat.AssemblyAngularVelocity.Y
			local targetYaw = -dir * CONFIG.TurnRate -- droite = rotation négative autour de Y
			local torque = (targetYaw - yawVel) * CONFIG.TurnKd
			seat:ApplyAngularImpulse(Vector3.yAxis * torque * seat.AssemblyMass * dt * CONFIG.TurnStrength)
		end
	end
end)

---------------------------------------------------------------------
-- Auto Wheelie + plateforme qui suit
---------------------------------------------------------------------
-- Cherche le ProximityPrompt (touche E) qui sert à monter sur la trott
local function findMountPrompt(seat)
	local container = seat:FindFirstAncestorOfClass("Model") or seat
	local fallback
	for _, d in ipairs(container:GetDescendants()) do
		if d:IsA("ProximityPrompt") and d.Enabled then
			if d.KeyboardKeyCode == Enum.KeyCode.E then
				return d
			end
			fallback = fallback or d
		end
	end
	return fallback
end

local lastReseat = 0
local ejectedAt = nil -- moment où tu as été éjecté
local needRecenter = false -- true = recentrer la trott dès que tu es de nouveau dessus
local lastJump = 0
-- Auto remonter : si tu es éjecté, tu es remis sur la trott
RunService.Heartbeat:Connect(function()
	local seat = getSeat()
	if seat then
		lastSeat = seat
		ejectedAt = nil
		if needRecenter then
			needRecenter = false
			if inZone then resetToStart(seat) end -- trott remise au centre, d'aplomb, après la remontée
		end
		return
	end
	if not ejectedAt then ejectedAt = os.clock() end
	if not state.AutoReseat or not lastSeat or not lastSeat.Parent then return end
	if not (state.AutoRide or state.AutoWheelie or state.AutoTurn or state.AutoLoop) then return end

	-- Ragdoll : un saut peut suffire pour se relever, on le tente en boucle pendant l'attente
	if CONFIG.AutoJumpOnEject and os.clock() - lastJump > CONFIG.JumpInterval and os.clock() - lastReseat > 2 then
		local c = player.Character
		local h = c and c:FindFirstChildOfClass("Humanoid")
		if h and h.Health > 0 then
			lastJump = os.clock()
			h.Jump = true
			pcall(function() h:ChangeState(Enum.HumanoidStateType.Jumping) end)
		end
	end

	if os.clock() - ejectedAt < CONFIG.ReseatDelay then return end -- attend la fin du ragdoll
	if os.clock() - lastReseat < 1 then return end

	local char = player.Character
	local hum = char and char:FindFirstChildOfClass("Humanoid")
	local root = char and char:FindFirstChild("HumanoidRootPart")
	if not hum or not root or hum.Health <= 0 then return end

	lastReseat = os.clock()
	needRecenter = true

	-- Dans la zone : on remet aussi la trott d'aplomb au départ (elle a pu se retourner)
	if inZone then
		resetToStart(lastSeat)
	end

	local prompt = CONFIG.UseMountPrompt and findMountPrompt(lastSeat)
	if prompt then
		-- Se place contre le prompt, puis simule un appui maintenu sur E
		local parent = prompt.Parent
		local pos = parent:IsA("Attachment") and parent.WorldPosition
			or (parent:IsA("BasePart") and parent.Position)
			or lastSeat.Position
		root.CFrame = CFrame.new(pos + Vector3.new(0, 2, 2))
		task.spawn(function()
			task.wait(0.2)
			pcall(function() prompt:InputHoldBegin() end)
			task.wait(prompt.HoldDuration + 0.15)
			pcall(function() prompt:InputHoldEnd() end)
		end)
	else
		-- Pas de prompt trouvé : on essaie de s'asseoir directement
		root.CFrame = lastSeat.CFrame * CFrame.new(0, 2, 0)
		pcall(function() lastSeat:Sit(hum) end)
		hum.Sit = true
	end
end)

RunService.Heartbeat:Connect(function(dt)
	local seat = getSeat()
	if not seat then return end

	-- Boucle auto : trop loin (ou tombé) -> retour au départ
	if state.AutoLoop and inZone and os.clock() - lastReset > 0.5 then
		local p = seat.Position
		local dx, dz = p.X - CONFIG.ZonePosition.X, p.Z - CONFIG.ZonePosition.Z
		if math.sqrt(dx * dx + dz * dz) > CONFIG.LoopDistance or p.Y < CONFIG.ZonePosition.Y - 20 then
			resetToStart()
			return
		end
	end


	if state.AutoWheelie then
		local look = seat.CFrame.LookVector
		local right = seat.CFrame.RightVector
		local pitch = math.deg(math.asin(math.clamp(look.Y, -1, 1)))
		local err = CONFIG.WheelieAngle - pitch
		local angVel = math.deg(seat.AssemblyAngularVelocity:Dot(right))
		local torque = CONFIG.WheelieKp * err - CONFIG.WheelieKd * angVel
		seat:ApplyAngularImpulse(right * torque * seat.AssemblyMass * dt * CONFIG.WheelieStrength)
	end

	if inZone and zone then
		local p = seat.Position
		local c = zone.Position
		local dx, dz = p.X - c.X, p.Z - c.Z
		if math.sqrt(dx * dx + dz * dz) > CONFIG.RecenterDistance then
			zone.Position = Vector3.new(p.X, c.Y, p.Z)
		end
	end
end)

player.CharacterAdded:Connect(function()
	inZone = false
	circleCenter = nil
	lastSeat = nil
	ejectedAt = nil
	needRecenter = false
	state.AutoRide = false
	state.AutoWheelie = false
	state.AutoTurn = false
	state.AutoLoop = false
	if _G.KukirinRefresh then _G.KukirinRefresh() end
end)

local refreshers = {}
local order = 1

local function makeToggle(label, key, onChange)
	local btn = Instance.new("TextButton")
	btn.Size = UDim2.new(1, -20, 0, 32)
	btn.Font = Enum.Font.Gotham
	btn.TextSize = 14
	btn.TextColor3 = Color3.new(1, 1, 1)
	btn.BorderSizePixel = 0
	btn.LayoutOrder = order
	order += 1
	btn.Parent = frame
	Instance.new("UICorner", btn).CornerRadius = UDim.new(0, 8)

	local function refresh()
		btn.Text = label .. " : " .. (state[key] and "ON" or "OFF")
		btn.BackgroundColor3 = state[key] and Color3.fromRGB(40, 160, 90) or Color3.fromRGB(70, 70, 80)
	end
	refreshers[key] = refresh

	btn.MouseButton1Click:Connect(function()
		status.Text = ""
		state[key] = not state[key]
		if onChange then onChange(state[key]) end
		refresh()
	end)
	refresh()
end

_G.KukirinRefresh = function()
	for _, f in pairs(refreshers) do f() end
end

makeToggle("Auto Ride", "AutoRide", function(on)
	if not on and not state.AutoTurn then
		local seat = getSeat()
		if seat and seat:IsA("VehicleSeat") then seat.ThrottleFloat = 0 end
	end
end)

makeToggle("Auto Wheelie", "AutoWheelie", function(on)
	if on then
		if CONFIG.TeleportOnWheelie and not inZone and not teleportToZone() then
			state.AutoWheelie = false
			status.Text = "Monte d'abord sur ton véhicule."
		end
	elseif not state.AutoTurn and not state.AutoLoop then
		returnFromZone()
	end
end)

makeToggle("Auto Tourner", "AutoTurn", function(on)
	local seat = getSeat()
	if on then
		if not seat then
			state.AutoTurn = false
			status.Text = "Monte d'abord sur ton véhicule."
			return
		end
		-- Le cercle est centré sur la zone de test si on y est, sinon sur la position actuelle
		if CONFIG.TeleportOnTurn and not inZone then
			teleportToZone()
		end
		circleCenter = inZone and CONFIG.ZonePosition or seat.Position
	else
		circleCenter = nil
		if not state.AutoWheelie and not state.AutoLoop then returnFromZone() end
		if seat and seat:IsA("VehicleSeat") then
			seat.SteerFloat = 0
			if not state.AutoRide then seat.ThrottleFloat = 0 end
		end
	end
end)

makeToggle("Boucle Auto", "AutoLoop", function(on)
	if on then
		if not inZone and not teleportToZone() then
			state.AutoLoop = false
			status.Text = "Monte d'abord sur ton véhicule."
		end
	elseif not state.AutoWheelie and not state.AutoTurn then
		returnFromZone()
	end
end)

local function makeSpeedButton(text, xScale, delta)
	local b = Instance.new("TextButton")
	b.Size = UDim2.new(0.22, 0, 1, 0)
	b.Position = UDim2.new(xScale, 0, 0, 0)
	b.Text = text
	b.Font = Enum.Font.GothamBold
	b.TextSize = 18
	b.TextColor3 = Color3.new(1, 1, 1)
	b.BackgroundColor3 = Color3.fromRGB(70, 70, 80)
	b.BorderSizePixel = 0
	b.Parent = speedRow
	Instance.new("UICorner", b).CornerRadius = UDim.new(0, 8)
	b.MouseButton1Click:Connect(function()
		state.TargetSpeed = math.clamp(state.TargetSpeed + delta, CONFIG.SpeedMin, CONFIG.SpeedMax)
		refreshSpeed()
	end)
end

UserInputService.InputBegan:Connect(function(input, processed)
	if processed then return end
	if input.KeyCode == CONFIG.ToggleKey then
		frame.Visible = not frame.Visible
	elseif input.KeyCode == CONFIG.ResetKey then
		if inZone then resetToStart() end
	end
end)
